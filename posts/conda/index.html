Conda 虚拟环境命令

📦 创建环境

# 创建一个新环境
conda create -n myenv python=3.10

# 创建环境并安装指定包
conda create -n myenv python=3.10 numpy pandas

🚀 激活与退出环境

# 激活环境
conda activate myenv

# 退出当前环境
conda deactivate

🔍 查看环境

# 查看所有虚拟环境
conda env list

🧹 删除环境

# 删除整个环境
conda remove -n myenv --all

📥 安装 / 卸载包

# 安装包
conda install numpy

# 安装指定版本的包
conda install numpy=1.26

# 卸载包
conda remove numpy

🔄 更新操作

# 更新某个包
conda update numpy

# 更新 conda 自身
conda update conda

# 更新环境中所有包
conda update --all

📄 导出与导入环境

# 导出环境配置到文件（包含 channel 信息）
conda env export > environment.yml

# 根据配置文件重新创建环境
conda env create -f environment.yml

🧰 其他实用命令

# 克隆一个已有环境
conda create --name newenv --clone oldenv

# 查看当前环境中的包
conda list
